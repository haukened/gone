{{- /* A Markdown table in a wrapper that scrolls sideways. Each cell carries its column's header (data-label), so on phones a row can stack as label and value. Code in cells may break after commas, so value lists such as 5m,30m,1h wrap between values, never inside one. Alignment is a class, not a style attribute, for the CSP. */ -}} {{- $heads := slice -}} {{- range .THead }}{{ range . }}{{ $heads = $heads | append (.Text | plainify | htmlUnescape) }}{{ end }}{{ end -}}
{{- range .THead }} {{ range . }}{{ .Text }}{{ end }} {{- end }} {{- range .TBody }} {{- /* A row whose first cell is one identifier (a GONE_* variable) or a flag gets a key; split-title.html turns it into an id, so the row can be linked and found by search. */}} {{- $first := (index . 0).Text | string -}} {{- $key := "" }}{{ $flag := "" -}} {{- with findRESubmatch `^([A-Z][A-Z0-9_]{2,})$` $first 1 }}{{ $key = index (index . 0) 1 | lower }}{{ end -}} {{- with findRESubmatch `^(?:-\w, )?--([\w-]+)[^<]*$` $first 1 }}{{ $flag = index (index . 0) 1 }}{{ end }} {{ range $c, $cell := . }}{{ $text := string $cell.Text }}{{ range seq 8 }}{{ $text = replaceRE `((?:[^<]|)*?),([^<])` "${1},${2}" $text }}{{ end }}{{ $text | safeHTML }}{{ end }} {{- end }}
