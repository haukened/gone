{{- /* A Markdown table in a wrapper that scrolls sideways. Each cell carries its column's header (data-label), so on phones a row can stack as label and value. Code in cells may break after commas, so value lists such as 5m,30m,1h wrap between values, never inside one. Alignment is a class, not a style attribute, for the CSP. */ -}} {{- $heads := slice -}} {{- range .THead }}{{ range . }}{{ $heads = $heads | append (.Text | plainify | htmlUnescape) }}{{ end }}{{ end -}}
{{- range .THead }} {{ range . }}{{ .Text }}{{ end }} {{- end }} {{- range .TBody }} {{ range $c, $cell := . }}{{ $text := string $cell.Text }}{{ range seq 8 }}{{ $text = replaceRE `((?:[^<]|)*?),([^<])` "${1},${2}" $text }}{{ end }}{{ $text | safeHTML }}{{ end }} {{- end }}
